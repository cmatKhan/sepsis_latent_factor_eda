# cwd is inside a read-only, mounted squashfs archive (see R/README.md's
# squashfs section) -- R's default graphics device otherwise lazily tries
# to create "Rplots.pdf" in cwd the first time ANY plotting call happens
# (even indirectly, inside a package internal to a job function), which
# fails hard there. Pre-opening a device pointed at a writable tempdir
# path avoids that regardless of what triggers it -- harmless if nothing
# ever plots.
grDevices::pdf(file.path(tempdir(), "Rplots.pdf"))

.rslurm_func <- readRDS('f.RDS')
.rslurm_params <- readRDS('params.RDS')
.rslurm_more_args <- readRDS('more_args.RDS')
.rslurm_id <- as.numeric(Sys.getenv('SLURM_ARRAY_TASK_ID'))
.rslurm_istart <- .rslurm_id * {{{nchunk}}} + 1
.rslurm_iend <- min((.rslurm_id + 1) * {{{nchunk}}}, nrow(.rslurm_params))
.rslurm_result <- do.call(parallel::mcmapply, c(
    FUN = .rslurm_func,
    .rslurm_params[.rslurm_istart:.rslurm_iend, , drop = FALSE],
    MoreArgs = list(.rslurm_more_args),
    mc.cores = {{{processes_per_node}}},
    mc.preschedule = {{{preschedule_cores}}},
    SIMPLIFY = FALSE))

.rslurm_output_dir <- Sys.getenv("RSLURM_OUTPUT_DIR", unset = ".")
dir.create(.rslurm_output_dir, recursive = TRUE, showWarnings = FALSE)
saveRDS(.rslurm_result, file = file.path(.rslurm_output_dir, paste0('results_', .rslurm_id, '.RDS')))
